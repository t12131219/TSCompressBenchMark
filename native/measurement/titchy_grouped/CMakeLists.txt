cmake_minimum_required(VERSION 3.22)
project(tsbench_titchy_grouped LANGUAGES CXX)
get_filename_component(repo "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(TSBENCH_FROZEN_TITCHY "" CACHE PATH "r40 Titchy C11 library and source header directory")
if(NOT IS_DIRECTORY "${TSBENCH_FROZEN_TITCHY}")
  message(FATAL_ERROR "Set TSBENCH_FROZEN_TITCHY to the r40 codec directory")
endif()
set(lib "${TSBENCH_FROZEN_TITCHY}/libtitchy_c.so")
set(headers "${TSBENCH_FROZEN_TITCHY}/source/titchy-native/include")
file(SHA256 "${lib}" library_hash)
file(SHA256 "${headers}/titchy.h" header_hash)
if(NOT library_hash STREQUAL "7ca1095ec4f8e28d7be131f436c4d2fc96faba40a1024c4441da596c2d3f917c" OR
   NOT header_hash STREQUAL "8e754a1635e1661af00ab98d03d3df166fdf6fc6227099985c9c379a35f61269")
  message(FATAL_ERROR "The grouped tool requires the r40 frozen Titchy C11 library and header")
endif()
add_library(titchy_frozen SHARED IMPORTED)
set_target_properties(titchy_frozen PROPERTIES IMPORTED_LOCATION "${lib}")
# Give the exact existing, single-unit controls distinct entry points without
# modifying their hashes or closing any wrapper/source difference by assumption.
set(base "${repo}/plugins/codecs/titchy-batch")
set(stream "${base}/corpus_stream_control.cpp")
set(access "${base}/corpus_access_control.cpp")
set_source_files_properties("${stream}" PROPERTIES COMPILE_DEFINITIONS "main=tsbench_titchy_stream_unit")
set_source_files_properties("${access}" PROPERTIES COMPILE_DEFINITIONS "main=tsbench_titchy_access_unit")
add_executable(tsbench_titchy_grouped main.cpp "${stream}" "${access}")
target_compile_features(tsbench_titchy_grouped PRIVATE cxx_std_17)
target_include_directories(tsbench_titchy_grouped PRIVATE "${headers}")
target_link_libraries(tsbench_titchy_grouped PRIVATE titchy_frozen)
set_target_properties(tsbench_titchy_grouped PROPERTIES CXX_EXTENSIONS NO BUILD_WITH_INSTALL_RPATH YES INSTALL_RPATH "$ORIGIN")
