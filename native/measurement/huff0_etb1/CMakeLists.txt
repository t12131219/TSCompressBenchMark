cmake_minimum_required(VERSION 3.22)
project(tsbench_huff0_etb1_diagnostic LANGUAGES CXX)

set(TSBENCH_FROZEN_HUFF0 "" CACHE PATH "Path to frozen Huff0 codec build directory")
set(headers "${TSBENCH_FROZEN_HUFF0}/source/FiniteStateEntropy/include")
set(lib "${TSBENCH_FROZEN_HUFF0}/libreference.so")
if(NOT EXISTS "${headers}/entropy_port.hpp" OR NOT EXISTS "${lib}")
  message(FATAL_ERROR "Set TSBENCH_FROZEN_HUFF0 to the frozen codec directory")
endif()
add_library(huff0_frozen SHARED IMPORTED)
set_target_properties(huff0_frozen PROPERTIES IMPORTED_LOCATION "${lib}")
add_executable(tsbench_huff0_etb1_timing main.cpp)
target_include_directories(tsbench_huff0_etb1_timing PRIVATE "${headers}")
target_link_libraries(tsbench_huff0_etb1_timing PRIVATE huff0_frozen)
target_compile_features(tsbench_huff0_etb1_timing PRIVATE cxx_std_17)
set_target_properties(tsbench_huff0_etb1_timing PROPERTIES CXX_EXTENSIONS NO BUILD_WITH_INSTALL_RPATH YES INSTALL_RPATH "$ORIGIN")
