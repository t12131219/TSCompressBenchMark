cmake_minimum_required(VERSION 3.22)
project(tsbench_lzbench_corpus_clock LANGUAGES CXX)
set(TSBENCH_FROZEN_LZBENCH "" CACHE PATH "Directory containing the frozen Unique lzbench native library")
set(source "${TSBENCH_FROZEN_LZBENCH}/source/lzbench-native")
set(lib "${TSBENCH_FROZEN_LZBENCH}/liblzbench_native.so")
set(xz_api "${source}/vendor/lzbench/lz/xz/src/liblzma/api")
if(NOT EXISTS "${source}/include/lzbench_extended.hpp" OR NOT EXISTS "${xz_api}/lzma.h" OR NOT EXISTS "${lib}")
  message(FATAL_ERROR "Set TSBENCH_FROZEN_LZBENCH to the frozen codec directory")
endif()
add_library(lzbench_frozen SHARED IMPORTED)
set_target_properties(lzbench_frozen PROPERTIES IMPORTED_LOCATION "${lib}")
foreach(family IN ITEMS session workers)
  set(target tsbench_lzbench_${family}_clock)
  add_executable(${target} ${family}.cpp)
  target_include_directories(${target} PRIVATE "${source}/include" "${xz_api}")
  target_link_libraries(${target} PRIVATE lzbench_frozen)
  target_compile_features(${target} PRIVATE cxx_std_17)
  set_target_properties(${target} PROPERTIES CXX_EXTENSIONS NO BUILD_WITH_INSTALL_RPATH YES INSTALL_RPATH "$ORIGIN")
endforeach()
# As with the frozen worker control: the shared library itself does not export
# lzma_stream_decoder. Resolve it from the same vendored static XZ kernels.
set(kernels "${TSBENCH_FROZEN_LZBENCH}/upstream/libln_kernels.a")
if(NOT EXISTS "${kernels}")
  message(FATAL_ERROR "Missing the frozen build's vendored kernels archive")
endif()
target_link_libraries(tsbench_lzbench_workers_clock PRIVATE "${kernels}" m)
