cmake_minimum_required(VERSION 3.22)
project(tsbench_corad_corpus_clock LANGUAGES C)
set(TSBENCH_CORAD_LIBRARY "" CACHE FILEPATH "Frozen M3 libcorad_c.so")
set(TSBENCH_CORAD_INCLUDE "" CACHE PATH "Frozen Unique CORAD headers")
if(NOT EXISTS "${TSBENCH_CORAD_LIBRARY}" OR NOT EXISTS "${TSBENCH_CORAD_INCLUDE}/corad.h")
  message(FATAL_ERROR "Bind the frozen CORAD library and header directory")
endif()
add_library(corad_frozen SHARED IMPORTED)
set_target_properties(corad_frozen PROPERTIES IMPORTED_LOCATION "${TSBENCH_CORAD_LIBRARY}")
add_executable(tsbench_corad_corpus_clock main.c)
target_include_directories(tsbench_corad_corpus_clock PRIVATE "${TSBENCH_CORAD_INCLUDE}")
target_link_libraries(tsbench_corad_corpus_clock PRIVATE corad_frozen)
target_compile_features(tsbench_corad_corpus_clock PRIVATE c_std_11)
set_target_properties(tsbench_corad_corpus_clock PROPERTIES C_EXTENSIONS NO BUILD_WITH_INSTALL_RPATH YES INSTALL_RPATH "$ORIGIN")
