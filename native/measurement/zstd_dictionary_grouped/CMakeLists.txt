cmake_minimum_required(VERSION 3.22)
project(tsbench_zstd_dictionary_grouped LANGUAGES CXX)
set(TSBENCH_FROZEN_LZBENCH "" CACHE PATH "Directory of the r40 liblzbench_native.so and source headers")
if(NOT IS_DIRECTORY "${TSBENCH_FROZEN_LZBENCH}")
  message(FATAL_ERROR "Set TSBENCH_FROZEN_LZBENCH to the r40 frozen codec directory")
endif()
set(source "${TSBENCH_FROZEN_LZBENCH}/source/lzbench-native")
set(lib "${TSBENCH_FROZEN_LZBENCH}/liblzbench_native.so")
function(check_frozen path expected)
  file(SHA256 "${path}" actual)
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "Zstd dictionary grouped harness requires the r40 identity: ${path}")
  endif()
endfunction()
check_frozen("${lib}" c5db81cbe6167151c2d82775f3a3bd0c8244663f892078436fec2d31aa5cbe08)
check_frozen("${source}/tools/feature_io.hpp" 9736361d8950495042fcc4ccc8d9a84dd85c5d23b9cf938391c095e6fa832762)
check_frozen("${source}/include/lzbench_extended.hpp" 8646662df4696b13e2304071ae71056fa826c521fca523e7e039b9134a251679)
check_frozen("${source}/include/lzbench_native.h" 02981e0a1db43365734053a9e6aa0af55675f8efd9edc4157e22c7b1e311bf19)
add_library(lzbench_frozen SHARED IMPORTED)
set_target_properties(lzbench_frozen PROPERTIES IMPORTED_LOCATION "${lib}")
add_executable(tsbench_zstd_dictionary_grouped main.cpp)
target_compile_features(tsbench_zstd_dictionary_grouped PRIVATE cxx_std_17)
target_include_directories(tsbench_zstd_dictionary_grouped PRIVATE "${source}/include" "${source}/tools")
target_link_libraries(tsbench_zstd_dictionary_grouped PRIVATE lzbench_frozen)
set_target_properties(tsbench_zstd_dictionary_grouped PROPERTIES CXX_EXTENSIONS NO BUILD_WITH_INSTALL_RPATH YES INSTALL_RPATH "$ORIGIN")
