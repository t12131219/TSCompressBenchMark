cmake_minimum_required(VERSION 3.22)
project(tsbench_pbepla_grouped LANGUAGES CXX)
get_filename_component(repo "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(TSBENCH_FROZEN_PBEPLA "" CACHE PATH "r40 PBEPLA-RR library and header directory")
if(NOT IS_DIRECTORY "${TSBENCH_FROZEN_PBEPLA}")
  message(FATAL_ERROR "Set TSBENCH_FROZEN_PBEPLA to the r40 codec directory")
endif()
set(lib "${TSBENCH_FROZEN_PBEPLA}/libtba_native.so")
set(headers "${TSBENCH_FROZEN_PBEPLA}/source/tsbenchmark-algorithms-native/include")
file(SHA256 "${lib}" library_hash)
file(SHA256 "${headers}/tba.h" header_hash)
if(NOT library_hash STREQUAL "271064f4dd7cf4759812c66ada33c5439ba89e2877dd91179b45453570f8a26c" OR
   NOT header_hash STREQUAL "e916ddad60d6ec15a656b696bb2ec633a569c5754764926ea0e50dff24d0582d")
  message(FATAL_ERROR "Requires the r40 frozen PBEPLA-RR build and header")
endif()
add_library(tba_frozen SHARED IMPORTED)
set_target_properties(tba_frozen PROPERTIES IMPORTED_LOCATION "${lib}")
set(old "${repo}/plugins/codecs/pbepla-rr/corpus_control.cpp")
set_source_files_properties("${old}" PROPERTIES COMPILE_DEFINITIONS "main=tsbench_pbepla_corpus_unit")
add_executable(tsbench_pbepla_grouped main.cpp "${old}")
target_include_directories(tsbench_pbepla_grouped PRIVATE "${headers}")
target_compile_features(tsbench_pbepla_grouped PRIVATE cxx_std_17)
target_compile_options(tsbench_pbepla_grouped PRIVATE -ffp-contract=off -fno-fast-math)
target_link_libraries(tsbench_pbepla_grouped PRIVATE tba_frozen ${CMAKE_DL_LIBS})
foreach(fn tba_pbepla_encode_ex tba_pbepla_encode_stream tba_pbepla_prefix tba_decode tba_pbepla_decode_stream tba_pbepla_inspect)
  target_link_options(tsbench_pbepla_grouped PRIVATE "LINKER:--wrap=${fn}")
endforeach()
set_target_properties(tsbench_pbepla_grouped PROPERTIES CXX_EXTENSIONS NO ENABLE_EXPORTS YES
  BUILD_WITH_INSTALL_RPATH YES INSTALL_RPATH "$ORIGIN")
