cmake_minimum_required(VERSION 3.22)
project(tsbench_selfstar_corpus_timing LANGUAGES CXX)
get_filename_component(repo "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(TSBENCH_FROZEN_SELFSTAR "" CACHE PATH "Directory containing the frozen SElf* native library")
set(headers "${TSBENCH_FROZEN_SELFSTAR}/source/elf-selfstar-native/include")
set(lib "${TSBENCH_FROZEN_SELFSTAR}/libselfstar_c.so")
if(NOT EXISTS "${headers}/selfstar.h" OR NOT EXISTS "${lib}")
  message(FATAL_ERROR "Set TSBENCH_FROZEN_SELFSTAR to the frozen codec directory")
endif()
add_library(selfstar_frozen SHARED IMPORTED)
set_target_properties(selfstar_frozen PROPERTIES IMPORTED_LOCATION "${lib}")
add_executable(tsbench_selfstar_timing main.cpp "${repo}/native/runtime/plugin_loader.cpp")
target_include_directories(tsbench_selfstar_timing PRIVATE "${headers}" "${repo}/include" "${repo}/native/runtime")
target_link_libraries(tsbench_selfstar_timing PRIVATE selfstar_frozen ${CMAKE_DL_LIBS})
target_compile_features(tsbench_selfstar_timing PRIVATE cxx_std_17)
set_target_properties(tsbench_selfstar_timing PROPERTIES CXX_EXTENSIONS NO BUILD_WITH_INSTALL_RPATH YES INSTALL_RPATH "$ORIGIN")
