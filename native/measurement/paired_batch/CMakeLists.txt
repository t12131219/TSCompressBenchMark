cmake_minimum_required(VERSION 3.22)
project(tsbench_paired_batch_workers LANGUAGES CXX)

# A separate development project: bind the existing codec build without
# recompiling/overwriting it or changing historical plugin CMake inputs.
get_filename_component(repo "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(TSBENCH_CODEC_BUILD "" CACHE PATH "Existing TSBench codec build to link against")
if(NOT IS_DIRECTORY "${TSBENCH_CODEC_BUILD}" OR TSBENCH_CODEC_BUILD STREQUAL CMAKE_BINARY_DIR)
  message(FATAL_ERROR "Set TSBENCH_CODEC_BUILD to a separate TSBench codec build")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED YES)
set(CMAKE_CXX_EXTENSIONS NO)
set(CMAKE_BUILD_WITH_INSTALL_RPATH YES)
set(CMAKE_INSTALL_RPATH "$ORIGIN")

add_library(tsbench_lane_loader STATIC "${repo}/native/runtime/plugin_loader.cpp")
target_include_directories(tsbench_lane_loader PUBLIC "${repo}/native/runtime" "${repo}/include")
target_link_libraries(tsbench_lane_loader PRIVATE ${CMAKE_DL_LIBS})
set(batch "${repo}/native/measurement/frame_batches.cpp")
add_executable(tsbench_frame_batch_dlopen "${batch}")
target_link_libraries(tsbench_frame_batch_dlopen PRIVATE tsbench_lane_loader)

function(linked_batch plugin package)
  set(module "${TSBENCH_CODEC_BUILD}/plugins/codecs/${package}/libbmk_${plugin}.so")
  if(NOT EXISTS "${module}")
    message(FATAL_ERROR "Missing codec library: ${module}")
  endif()
  add_library(tsbench_lane_codec_${plugin} SHARED IMPORTED)
  set_property(TARGET tsbench_lane_codec_${plugin} PROPERTY IMPORTED_LOCATION "${module}")
  set(target "tsbench_frame_batch_tsbench_${plugin}")
  add_executable(${target} "${batch}")
  target_compile_definitions(${target} PRIVATE TSBENCH_BATCH_LINKED=1)
  target_link_libraries(${target} PRIVATE tsbench_lane_loader tsbench_lane_codec_${plugin})
  target_link_options(${target} PRIVATE "LINKER:-z,now")
endfunction()

linked_batch(alp_auto alp-auto)
linked_batch(pcodec_source_auto pcodec)
linked_batch(pcodec_c_native_auto pcodec)
linked_batch(zstd zstd)
linked_batch(xz_baseline zstd)
