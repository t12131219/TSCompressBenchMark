cmake_minimum_required(VERSION 3.22)
project(tsbench VERSION 0.1.0 LANGUAGES C CXX)

add_library(tsbench_plugin_abi INTERFACE)
target_include_directories(tsbench_plugin_abi INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(tsbench_plugin_abi INTERFACE c_std_11 cxx_std_17)

include(CTest)
if(BUILD_TESTING)
  add_executable(tsbench_abi_c11 tests/contracts/abi_contract.c)
  target_link_libraries(tsbench_abi_c11 PRIVATE tsbench_plugin_abi)
  set_target_properties(tsbench_abi_c11 PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
  add_executable(tsbench_abi_cpp17 tests/contracts/abi_contract.cpp)
  target_link_libraries(tsbench_abi_cpp17 PRIVATE tsbench_plugin_abi)
  set_target_properties(tsbench_abi_cpp17 PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
  add_test(NAME tsbench.abi.c11 COMMAND tsbench_abi_c11)
  add_test(NAME tsbench.abi.cpp17 COMMAND tsbench_abi_cpp17)
endif()

if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR STREQUAL "x86_64")
  find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
  include(cmake/frozen.cmake)
  include(cmake/plugins.cmake)
  tsbench_add_registered_plugins()
  option(TSBENCH_PROFILE_BUILD "Build separate ALP/Zstd native trace overlays for M4 diagnostics" OFF)
  include(cmake/trace-profile.cmake)
  if(TSBENCH_PROFILE_BUILD)
    # Do not provide normal execution/measurement workers in an overlay tree.
    return()
  endif()
  include(cmake/m3-controls.cmake)
  include(cmake/pbepla-corpus.cmake)
  include(cmake/lzbench-corpus.cmake)
  include(cmake/lzbench-workers.cmake)
  include(cmake/huff0-corpus.cmake)
  add_library(tsbench_plugin_loader STATIC native/runtime/plugin_loader.cpp)
  target_link_libraries(tsbench_plugin_loader PUBLIC tsbench_plugin_abi PRIVATE ${CMAKE_DL_LIBS})
  target_include_directories(tsbench_plugin_loader PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/native/runtime")
  set_target_properties(tsbench_plugin_loader PROPERTIES CXX_EXTENSIONS NO)
  include(cmake/huff0-fec1.cmake)
  add_executable(tsbench_cpu_worker native/worker/worker_main.cpp)
  target_link_libraries(tsbench_cpu_worker PRIVATE tsbench_plugin_loader)
  set_target_properties(tsbench_cpu_worker PROPERTIES CXX_EXTENSIONS NO)
  add_executable(tsbench_corpus_worker native/worker/corpus_worker.cpp)
  target_link_libraries(tsbench_corpus_worker PRIVATE tsbench_plugin_loader)
  set_target_properties(tsbench_corpus_worker PROPERTIES CXX_EXTENSIONS NO)
  add_executable(tsbench_tv_worker native/worker/tv_worker.cpp)
  target_include_directories(tsbench_tv_worker PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/native/pipeline")
  target_link_libraries(tsbench_tv_worker PRIVATE tsbench_plugin_loader)
  set_target_properties(tsbench_tv_worker PROPERTIES CXX_EXTENSIONS NO)
  if(BUILD_TESTING)
    include(cmake/development-tests.cmake)
  endif()
endif()
